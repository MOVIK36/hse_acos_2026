cmake_minimum_required(VERSION 3.10)
project(sort_center C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -pedantic)
find_package(Threads REQUIRED)
file(GLOB SOURCES "src/*.c")
add_executable(sort_center ${SOURCES})
target_link_libraries(sort_center PRIVATE Threads::Threads rt)
add_executable(sort_source src/clients/source.c src/tcp.c)
add_executable(sort_observer src/clients/observer.c src/tcp.c)
target_include_directories(sort_source PRIVATE src)
target_include_directories(sort_observer PRIVATE src)
